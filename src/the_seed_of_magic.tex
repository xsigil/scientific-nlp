\end{align}

% --- 因果ステージ軸（Stage-wise Decomposition） ---
% ステージ空間に対して、情動パラメータ (profit, threat, neutral) の直和として定義
\begin{align}
    \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{cause}}     &\equiv \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{cause}}(\mathrm{profit}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{cause}}(\mathrm{threat}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{cause}}(\mathrm{neutral}) \\
    \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{causality}} &\equiv \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{causality}}(\mathrm{profit}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{causality}}(\mathrm{threat}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{causality}}(\mathrm{neutral}) \\
    \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{effect}}    &\equiv \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{effect}}(\mathrm{profit}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{effect}}(\mathrm{threat}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{effect}}(\mathrm{neutral})
\end{align}

% --- 情動ドライブ軸（Drive-wise Decomposition） ---
% 各パラメータ空間が、3つの因果ステージの直和として構成される
\begin{align}
    \tilde{\Psi}^{\mathrm{ego}}(\mathrm{profit})  &\equiv \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{cause}}(\mathrm{profit}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{causality}}(\mathrm{profit}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{effect}}(\mathrm{profit}) \\
    \tilde{\Psi}^{\mathrm{ego}}(\mathrm{threat})  &\equiv \tilde{\Psi}^{\.ego}_{\mathrm{cause}}(\mathrm{threat}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{causality}}(\mathrm{threat}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{effect}}(\mathrm{threat}) \\
    \tilde{\Psi}^{\mathrm{ego}}(\mathrm{neutral}) &\equiv \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{cause}}(\mathrm{neutral}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{causality}}(\mathrm{neutral}) \cup \tilde{\Psi}^{\mathrm{ego}}_{\mathrm{effect}}(\mathrm{neutral})
\end{align}

\end{equation}
