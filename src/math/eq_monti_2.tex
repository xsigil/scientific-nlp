\begin{enumerate}[label=(\roman*)]
    \item $\mthEventC_1$（車がドア1にある）：残るドア2とドア3は共にハズレであるため、ランダムに開ける。
        \begin{equation}
            \mthConditionalProbability{\mthEventM_3}{\mthEventC_1} = \frac{1}{2}
        \end{equation}
    \item $\mthEventC_2$（車がドア2にある）：車があるドア2は開けられないため、必然的にドア3を開けるしかない。
        \begin{equation}
            \mthConditionalProbability{\mthEventM_3}{\mthEventC_2} = 1
        \end{equation}
    \item $\mthEventC_3$（車がドア3にある）：車があるドア3を開けることはルール上あり得ない。
        \begin{equation}
            \mthConditionalProbability{\mthEventM_3}{\mthEventC_3} = 0
        \end{equation}
\end{enumerate}
